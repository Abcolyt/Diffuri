# ============================================================================
# data/constants.txt
#
# Файл именованных констант проекта Diffuri.
#
# Читается парсером (модуль input/parser) при каждом запуске программы.
# Путь ищется относительно РАБОЧЕЙ ДИРЕКТОРИИ (CWD) процесса, а не
# относительно расположения экзешника. Это значит:
#   - при запуске из отладчика или из папки сборки файл найдётся
#     (т.к. копия лежит рядом с экзешником в подпапке data/);
#   - при запуске из произвольной папки запускайте экзешник так, чтобы
#     текущей директорией была папка с экзешником:
#       cd out/build/x64-debug
#       ./Diffuri.exe < system.txt
#
# ----------------------------------------------------------------------------
# ФОРМАТ
# ----------------------------------------------------------------------------
# Одна константа на строку: "имя = значение".
#   - Пустые строки игнорируются.
#   - Строки, начинающиеся с '#', игнорируются (комментарии).
#   - Всё, что стоит после '#' в строке с константой, обрезается.
#
# КАК ДОБАВИТЬ НОВУЮ КОНСТАНТУ
# ----------------------------------------------------------------------------
#   1. Впишите строку вида "имя = значение" в любое место файла
#      (удобнее всего — в соответствующую секцию ниже).
#   2. Имя должно начинаться с буквы или '_', далее буквы/цифры/'_':
#         правильно:   mu, g_earth, alpha_1, _internal
#         неправильно: 2mu, mu-x, mu.x
#   3. Значение — любое число в формате double. Допустимые формы:
#         3.14159
#         -2.5
#         1e-3          (экспоненциальная форма, строчная 'e')
#         2.5E+7        (экспоненциальная форма, заглавная 'E')
#         0x1.8p+1      — НЕ поддерживается (только десятичная запись)
#   4. Пересоберите проект: копия файла рядом с экзешником обновится
#      автоматически при следующей сборке.
#
# ПРАВИЛО РЕДАКТИРОВАНИЯ
# ----------------------------------------------------------------------------
# Источник правды — этот файл в корне проекта (в репозитории).
# Копия рядом с экзешником (например, в out/build/x64-debug/data/)
# ПЕРЕЗАПИСЫВАЕТСЯ при каждой сборке. Хотите поменять константу —
# редактируйте файл в репозитории, а не копию в папке сборки.
#
# ПРИОРИТЕТ
# ----------------------------------------------------------------------------
# Если имя задано одновременно здесь и программно через
# ParseOptions::extra_constants, приоритет у программного значения
# (см. input/parser.h, LoadConstants).
# ============================================================================

# --- Константы задач ---------------------------------------------------------
# CR3BP (круговая ограниченная задача трёх тел).
# Массовый параметр μ = m₂/(m₁+m₂) — отношение массы второго тела (Луны)
# к суммарной массе системы (Земля+Луна). Безразмерная величина.
#   Земля–Луна:    mu = 0.0121505856   (1−μ = 0.9878494144)
#   Солнце–Земля:  mu = 0.000003003    (1−μ = 0.999996997)
#   Солнце–Юпитер: mu = 0.000953875    (1−μ = 0.999046125)
mu = 0.0121505856

# Дополнительная константа 1−μ. Вводится отдельно, чтобы уравнения
# читались как в литературе: коэффициент при первом притягивающем центре
# равен (1−μ), при втором — μ. Для системы Земля–Луна:
one_minus_mu = 0.9878494144

# --- Полная задача трёх тел (неограниченная, 3D) -----------------------------
# Три тела с массами m1, m2, m3, взаимное гравитационное притяжение.
# Никаких ограничений: тела массивные, орбиты произвольные, движение в 3D.
#
# Единицы измерения — нормализованные (стандарт для классической задачи
# трёх тел, например, орбита «восьмёрка» Ченсинера–Монтгомери):
#   G = 1, массы в произвольных единицах, длины в тех же единицах.
# Определение G = 1.0 ниже ПЕРЕОПРЕДЕЛЯЕТ G = 6.67430e-11 (SI),
# заданное выше: парсер читает файл последовательно, последнее значение
# выигрывает. Если нужны СИ — уберите эту строку и используйте SI-значения
# масс (кг) и расстояний (м).
G = 1.0

# Массы трёх тел. Для классической задачи с равными массами — все по 1.
m1 = 1.0
m2 = 1.0
m3 = 1.0
